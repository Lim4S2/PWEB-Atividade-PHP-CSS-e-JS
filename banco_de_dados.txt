-- 1. Criação do Banco de Dados
CREATE DATABASE IF NOT EXISTS `carvotrack` 
  CHARACTER SET utf8mb4 
  COLLATE utf8mb4_unicode_ci;

USE `carvotrack`;

-- ---------------------------------------------------------------------
-- 2. Tabela de Usuários / Operadores
-- Armazena os usuários autorizados a acessar o sistema
-- ---------------------------------------------------------------------
DROP TABLE IF EXISTS `historico_acoes`;
DROP TABLE IF EXISTS `processos_carbonizacao`;
DROP TABLE IF EXISTS `lotes_madeira`;
DROP TABLE IF EXISTS `fornos`;
DROP TABLE IF EXISTS `usuarios`;

CREATE TABLE `usuarios` (
  `id` INT AUTO_INCREMENT PRIMARY KEY,
  `nome` VARCHAR(100) NOT NULL,
  `email` VARCHAR(120) NOT NULL UNIQUE,
  `senha` VARCHAR(255) NOT NULL,
  `cargo` VARCHAR(50) NOT NULL DEFAULT 'Operador',
  `criado_em` TIMESTAMP DEFAULT CURRENT_TIMESTAMP
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- ---------------------------------------------------------------------
-- 3. Tabela de Fornos de Carbonização (Entidade principal do CRUD)
-- Representa as fornalhas industriais monitoradas
-- ---------------------------------------------------------------------
CREATE TABLE `fornos` (
  `id` INT AUTO_INCREMENT PRIMARY KEY,
  `codigo` VARCHAR(30) NOT NULL UNIQUE,
  `tipo_material` VARCHAR(60) NOT NULL,
  `capacidade_kg` DECIMAL(10,2) NOT NULL,
  `potencia_kw` DECIMAL(8,2) NOT NULL,
  `temperatura_max` DECIMAL(6,2) NOT NULL,
  `status` ENUM('disponivel', 'em_processo', 'manutencao') NOT NULL DEFAULT 'disponivel',
  `observacoes` TEXT NULL,
  `criado_em` TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  `atualizado_em` TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- ---------------------------------------------------------------------
-- 4. Tabela de Lotes de Madeira
-- Lotes de biomassa/madeira para carbonização
-- ---------------------------------------------------------------------
CREATE TABLE `lotes_madeira` (
  `id` INT AUTO_INCREMENT PRIMARY KEY,
  `codigo_lote` VARCHAR(30) NOT NULL UNIQUE,
  `especie_madeira` VARCHAR(80) NOT NULL,
  `peso_lote_kg` DECIMAL(10,2) NOT NULL,
  `umidade_percentual` DECIMAL(5,2) NOT NULL,
  `densidade_kg_m3` DECIMAL(6,2) NOT NULL,
  `status` ENUM('disponivel', 'em_uso', 'concluido') NOT NULL DEFAULT 'disponivel',
  `data_entrada` DATE NOT NULL,
  `criado_em` TIMESTAMP DEFAULT CURRENT_TIMESTAMP
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- ---------------------------------------------------------------------
-- 5. Tabela de Processos de Carbonização (Escrita Multi-tabelas / Transação)
-- Vincula forno, lote de madeira e operador
-- ---------------------------------------------------------------------
CREATE TABLE `processos_carbonizacao` (
  `id` INT AUTO_INCREMENT PRIMARY KEY,
  `forno_id` INT NOT NULL,
  `lote_id` INT NOT NULL,
  `usuario_id` INT NOT NULL,
  `data_inicio` DATETIME NOT NULL,
  `data_fim` DATETIME NULL,
  `temperatura_alvo` DECIMAL(6,2) NOT NULL,
  `rendimento_carvao_kg` DECIMAL(10,2) NULL,
  `status` ENUM('ativo', 'finalizado', 'cancelado') NOT NULL DEFAULT 'ativo',
  `criado_em` TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  CONSTRAINT `fk_processo_forno` FOREIGN KEY (`forno_id`) REFERENCES `fornos` (`id`) ON UPDATE CASCADE ON DELETE RESTRICT,
  CONSTRAINT `fk_processo_lote` FOREIGN KEY (`lote_id`) REFERENCES `lotes_madeira` (`id`) ON UPDATE CASCADE ON DELETE RESTRICT,
  CONSTRAINT `fk_processo_usuario` FOREIGN KEY (`usuario_id`) REFERENCES `usuarios` (`id`) ON UPDATE CASCADE ON DELETE RESTRICT
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- ---------------------------------------------------------------------
-- 6. Tabela de Histórico de Ações (Auditoria na Transação)
-- ---------------------------------------------------------------------
CREATE TABLE `historico_acoes` (
  `id` INT AUTO_INCREMENT PRIMARY KEY,
  `usuario_id` INT NOT NULL,
  `acao` VARCHAR(80) NOT NULL,
  `detalhes` TEXT NULL,
  `criado_em` TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  CONSTRAINT `fk_historico_usuario` FOREIGN KEY (`usuario_id`) REFERENCES `usuarios` (`id`) ON UPDATE CASCADE ON DELETE RESTRICT
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;
